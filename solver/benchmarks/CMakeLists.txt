cmake_minimum_required(VERSION 3.20)

# Helper macro: add a standalone benchmark executable
macro(smf_benchmark name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE smf)
    if(TARGET Eigen3::Eigen)
        target_link_libraries(${name} PRIVATE Eigen3::Eigen)
        target_compile_definitions(${name} PRIVATE SMF_HAS_EIGEN=1)
    endif()
endmacro()

smf_benchmark(bench_poisson)
smf_benchmark(bench_kkt_ocp)
smf_benchmark(bench_random_symmetric)
smf_benchmark(bench_suite_sparse_matrix_market)
smf_benchmark(bench_compare)
smf_benchmark(bench_kkt_fixture_compare)

# bench_compare: link optional comparison solvers (CHOLMOD, MA27)
if(SMF_HAS_CHOLMOD AND CHOLMOD_LIBRARY AND CHOLMOD_INCLUDE_DIR)
    target_link_libraries(bench_compare PRIVATE ${CHOLMOD_LIBRARY})
    target_include_directories(bench_compare PRIVATE ${CHOLMOD_INCLUDE_DIR})
    target_compile_definitions(bench_compare PRIVATE SMF_HAS_CHOLMOD=1)
endif()
if(SMF_HAS_MA27 AND COINHSL_LIBRARY)
    target_link_libraries(bench_compare PRIVATE ${COINHSL_LIBRARY})
    target_compile_definitions(bench_compare PRIVATE SMF_HAS_MA27=1)
    target_link_libraries(bench_kkt_fixture_compare PRIVATE ${COINHSL_LIBRARY})
    target_compile_definitions(bench_kkt_fixture_compare PRIVATE SMF_HAS_MA27=1)
endif()
if(SMF_HAS_MUMPS AND DMUMPS_LIBRARY AND MUMPS_INCLUDE_DIR)
    set(_mumps_libs ${DMUMPS_LIBRARY})
    if(MUMPS_COMMON_LIBRARY)
        list(APPEND _mumps_libs ${MUMPS_COMMON_LIBRARY})
    endif()
    if(MPISEQ_LIBRARY)
        list(APPEND _mumps_libs ${MPISEQ_LIBRARY})
    endif()
    target_link_libraries(bench_compare PRIVATE ${_mumps_libs})
    target_include_directories(bench_compare PRIVATE ${MUMPS_INCLUDE_DIR})
    target_compile_definitions(bench_compare PRIVATE SMF_HAS_MUMPS=1)
endif()

# -------------------------------------------------------------------------
# bench_ipopt_compare: smf vs MA27 as IPOPT linear solvers
# Enabled by -DSMF_BUILD_IPOPT_BENCHMARK=ON (requires IPOPT installed)
# -------------------------------------------------------------------------
if(SMF_BUILD_IPOPT_BENCHMARK)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(IPOPT QUIET ipopt)
    endif()
    if(NOT IPOPT_FOUND)
        # Fallback: look in standard CoinOR paths
        find_library(IPOPT_LIBRARY NAMES ipopt PATHS /usr/local/lib /usr/lib)
        find_path(IPOPT_INCLUDE_DIR
            NAMES IpIpoptApplication.hpp
            PATHS /usr/local/include/coin-or /usr/include/coin-or /usr/include/coin)
        if(IPOPT_LIBRARY AND IPOPT_INCLUDE_DIR)
            set(IPOPT_FOUND TRUE)
            set(IPOPT_LIBRARIES ${IPOPT_LIBRARY})
            set(IPOPT_INCLUDE_DIRS ${IPOPT_INCLUDE_DIR})
        endif()
    endif()

    if(IPOPT_FOUND)
        add_executable(bench_ipopt_compare bench_ipopt_compare.cpp)
        target_link_libraries(bench_ipopt_compare PRIVATE smf ${IPOPT_LIBRARIES})
        target_include_directories(bench_ipopt_compare PRIVATE ${IPOPT_INCLUDE_DIRS})
        if(IPOPT_LIBRARY_DIRS)
            target_link_directories(bench_ipopt_compare PRIVATE ${IPOPT_LIBRARY_DIRS})
        endif()
        if(SMF_HAS_MA27 AND COINHSL_LIBRARY)
            target_link_libraries(bench_ipopt_compare PRIVATE ${COINHSL_LIBRARY})
        endif()
        message(STATUS "bench_ipopt_compare: enabled (IPOPT found at ${IPOPT_LIBRARIES})")
    else()
        message(STATUS "bench_ipopt_compare: disabled (IPOPT not found; set SMF_BUILD_IPOPT_BENCHMARK=OFF to suppress)")
    endif()
endif()
